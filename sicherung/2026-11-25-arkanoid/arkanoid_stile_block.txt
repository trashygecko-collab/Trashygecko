/* Jedes Extra bekommt ein eigenes Symbol aus Linien – keine Schrift, damit es
   in jeder Größe und auch im Fallen eindeutig bleibt. */
function powerSymbol(g,art,x,y){
 g.lineWidth=1.9;
 g.lineCap="round";
 g.lineJoin="round";
 g.beginPath();
 if(art==="E"){                       // Doppelpfeil nach außen
  g.moveTo(x-1.6,y); g.lineTo(x-6,y);
  g.moveTo(x-4.1,y-2.2); g.lineTo(x-6.1,y); g.lineTo(x-4.1,y+2.2);
  g.moveTo(x+1.6,y); g.lineTo(x+6,y);
  g.moveTo(x+4.1,y-2.2); g.lineTo(x+6.1,y); g.lineTo(x+4.1,y+2.2);
 }else if(art==="K"){                 // Doppelpfeil nach innen
  g.moveTo(x-6,y); g.lineTo(x-1.6,y);
  g.moveTo(x-3.7,y-2.2); g.lineTo(x-1.7,y); g.lineTo(x-3.7,y+2.2);
  g.moveTo(x+6,y); g.lineTo(x+1.6,y);
  g.moveTo(x+3.7,y-2.2); g.lineTo(x+1.7,y); g.lineTo(x+3.7,y+2.2);
 }else if(art==="D"){                 // drei Kugeln
  g.moveTo(x+2.3,y-3.6); g.arc(x,y-3.6,2.3,0,Math.PI*2);
  g.moveTo(x-1,y+3.1); g.arc(x-3.3,y+3.1,2.3,0,Math.PI*2);
  g.moveTo(x+5.6,y+3.1); g.arc(x+3.3,y+3.1,2.3,0,Math.PI*2);
 }else if(art==="L"){                 // zwei Laserstrahlen
  g.moveTo(x-3,y+5.2); g.lineTo(x-3,y-5.2);
  g.moveTo(x+3,y+5.2); g.lineTo(x+3,y-5.2);
  g.moveTo(x-4.8,y-3.2); g.lineTo(x-3,y-5.4); g.lineTo(x-1.2,y-3.2);
  g.moveTo(x+1.2,y-3.2); g.lineTo(x+3,y-5.4); g.lineTo(x+4.8,y-3.2);
 }else if(art==="C"){                 // Fangnetz
  g.rect(x-5.2,y-4.6,10.4,9.2);
  g.moveTo(x-5.2,y-1.5); g.lineTo(x+5.2,y-1.5);
  g.moveTo(x-5.2,y+1.5); g.lineTo(x+5.2,y+1.5);
  g.moveTo(x-1.8,y-4.6); g.lineTo(x-1.8,y+4.6);
  g.moveTo(x+1.8,y-4.6); g.lineTo(x+1.8,y+4.6);
 }else if(art==="T"){                 // Kugel mit Energiespur
  g.moveTo(x-6,y-2.5); g.lineTo(x-1.4,y-2.5);
  g.moveTo(x-6,y+2.5); g.lineTo(x-1.4,y+2.5);
  g.moveTo(x+4.9,y); g.arc(x+2.1,y,2.8,0,Math.PI*2);
 }else if(art==="G"){                 // Schild
  g.moveTo(x,y-5.8);
  g.lineTo(x+5,y-3.4); g.lineTo(x+4.4,y+1.8);
  g.quadraticCurveTo(x+2,y+5.4,x,y+5.9);
  g.quadraticCurveTo(x-2,y+5.4,x-4.4,y+1.8);
  g.lineTo(x-5,y-3.4); g.closePath();
 }else if(art==="P"){                 // Herz
  g.moveTo(x,y+4.8);
  g.bezierCurveTo(x-6.6,y+0.4,x-4.2,y-5.4,x,y-2.4);
  g.bezierCurveTo(x+4.2,y-5.4,x+6.6,y+0.4,x,y+4.8);
 }else if(art==="S"){                 // Sanduhr
  g.moveTo(x-4.4,y-5); g.lineTo(x+4.4,y-5);
  g.lineTo(x-4.4,y+5); g.lineTo(x+4.4,y+5); g.closePath();
  g.moveTo(x-2.6,y-5); g.lineTo(x+2.6,y+5);
 }else if(art==="F"){                 // Blitz
  g.moveTo(x+1.6,y-6); g.lineTo(x-3.4,y+0.6); g.lineTo(x+0.2,y+0.6);
  g.lineTo(x-1.6,y+6); g.lineTo(x+3.6,y-1.2); g.lineTo(x-0.2,y-1.2); g.closePath();
 }else if(art==="B"){                 // Stern
  for(let i=0;i<10;i++){
   const r=i%2?2.7:6.2, a=-Math.PI/2+i*Math.PI/5;
   if(i)g.lineTo(x+Math.cos(a)*r,y+Math.sin(a)*r);
   else g.moveTo(x+Math.cos(a)*r,y+Math.sin(a)*r);
  }
  g.closePath();
 }
 g.stroke();
}

/* Risse im Stein nach dem ersten Treffer eines Doppelsteins. */
function rissZeichnen(g,st,farbe){
 g.strokeStyle=farbe;
 g.lineWidth=1.1;
 g.beginPath();
 g.moveTo(st.x+st.w*0.30,st.y+st.h*0.10);
 g.lineTo(st.x+st.w*0.40,st.y+st.h*0.50);
 g.lineTo(st.x+st.w*0.31,st.y+st.h*0.90);
 g.moveTo(st.x+st.w*0.66,st.y+st.h*0.14);
 g.lineTo(st.x+st.w*0.57,st.y+st.h*0.56);
 g.lineTo(st.x+st.w*0.63,st.y+st.h*0.88);
 g.stroke();
}

/* Drei Ansichten, ein Spiel. „Klassisch“ bleibt die grün-goldene Fassung.
   „Mädchen“ ist Perlmutt und Kristall: Regenbogenschimmer, Goldkanten,
   Glitzer. „Jungs“ ist Chrom und Plasma: Stahl, Cyan, Lichtschweif.
   Jede Ansicht hat einen gecachten Hintergrund und eine kleine lebende
   Schicht (Glitzer bzw. Puls) – Maße, Punkte und Tempo sind überall gleich. */
function hell(hex,t){
 const n=parseInt(hex.slice(1),16), r=n>>16, gg=(n>>8)&255, b=n&255;
 return "rgb("+Math.round(r+(255-r)*t)+","+Math.round(gg+(255-gg)*t)+","+Math.round(b+(255-b)*t)+")";
}
function dunkel(hex,t){
 const n=parseInt(hex.slice(1),16), r=n>>16, gg=(n>>8)&255, b=n&255;
 return "rgb("+Math.round(r*(1-t))+","+Math.round(gg*(1-t))+","+Math.round(b*(1-t))+")";
}
function herz(g,x,y,s){
 g.beginPath();
 g.moveTo(x,y+s*0.6);
 g.bezierCurveTo(x-s*1.3,y-s*0.5,x-s*0.35,y-s*1.15,x,y-s*0.35);
 g.bezierCurveTo(x+s*0.35,y-s*1.15,x+s*1.3,y-s*0.5,x,y+s*0.6);
 g.closePath();
 g.fill();
}
